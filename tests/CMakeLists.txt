# ecs_add_test(<nome> <fontes...>)
function(ecs_add_test name)
  add_executable(${name} ${ARGN} ${CMAKE_CURRENT_SOURCE_DIR}/support/doctest_main.cpp)
  target_link_libraries(${name} PRIVATE ecs doctest::doctest)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/support)
  ecs_warnings(${name})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

ecs_add_test(unit_table unit_table.cpp)
ecs_add_test(unit_type_info unit_type_info.cpp)
ecs_add_test(unit_entity unit_entity.cpp)
ecs_add_test(unit_archetype unit_archetype.cpp)
ecs_add_test(unit_world unit_world.cpp support/alloc_counter.cpp)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bench)

# ecs_add_bench(<nome> <fontes...>)
function(ecs_add_bench name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE ecs benchmark::benchmark)
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  ecs_warnings(${name})
  if(ECS_VEC_REPORT)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(${name} PRIVATE -fopt-info-vec-optimized)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      target_compile_options(${name} PRIVATE -Rpass=loop-vectorize)
    endif()
  endif()
endfunction()

ecs_add_bench(bench_iterate iterate.cpp)
ecs_add_bench(bench_transitions transitions.cpp)
